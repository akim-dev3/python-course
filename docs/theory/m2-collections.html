Модуль 2 — Коллекции

Конспект обновлён 2026-06-09

Суть

Три основные коллекции Python: list, tuple, set. Полностью закрыто к 23.05.

Детали

List — изменяемая последовательность

lst = [1, 2, 3, 4, 5]

# Доступ
lst[0]      # 1 — первый
lst[-1]     # 5 — последний
lst[1:3]    # [2, 3] — срез (end не включается)
lst[::-1]   # [5, 4, 3, 2, 1] — реверс

# Изменение
lst.append(6)           # добавить в конец
lst.extend([7, 8])      # добавить несколько
lst.insert(0, 0)        # вставить на позицию
lst.remove(3)           # удалить первое вхождение
lst.pop()               # удалить последний (возвращает его)
lst.pop(0)              # удалить по индексу

# Информация
len(lst)                # длина
lst.count(2)            # сколько раз встречается
lst.index(4)            # индекс первого вхождения
4 in lst                # True/False — проверка

# Сортировка
lst.sort()                          # на месте (None)
lst.sort(reverse=True)
sorted(lst)                         # новый список
sorted(lst, key=lambda x: -x)      # по ключу

List Comprehension — главный инструмент

# [выражение for элемент in список if условие]

squares = [x ** 2 for x in range(10)]
evens = [x for x in lst if x % 2 == 0]
names = [p["name"] for p in players if p["active"]]

# Вложенный
flat = [item for row in matrix for item in row]

# С if/else в выражении — else обязателен, синтаксис другой
labels = ["adult" if a >= 18 else "minor" for a in ages]

Set и Dict Comprehension

# Set comprehension — фигурные скобки без двоеточия
unique_lens = {len(w) for w in words}

# Dict comprehension — фигурные с двоеточием
squares = {x: x ** 2 for x in range(5)}
# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}

# Инверсия dict
inverted = {v: k for k, v in d.items()}

# Фильтрация dict
adults = {name: age for name, age in users.items() if age >= 18}

Generator expression — () вместо []

# Не создаёт список целиком — итерирует по одному элементу
total = sum(x ** 2 for x in range(1_000_000))
any(x < 0 for x in lst)         # короткое замыкание — стоп при первом True

Используй когда нужна сумма/max/any/all, а сам список не нужен. Экономит память.

Tuple — неизменяемая последовательность

t = (1, 2, 3)
t = 1, 2, 3      # скобки не обязательны

# Доступ — как у list
t[0]    # 1
t[-1]   # 3

# Распаковка
x, y, z = t
first, *rest = t    # first=1, rest=[2,3]
a, b = b, a         # обмен значений

# Когда использовать tuple вместо list:
# - фиксированный набор значений (координаты, RGB, конфиг)
# - как ключ словаря (list нельзя — не hashable)
# - возврат нескольких значений из функции

Set — неупорядоченное множество уникальных элементов

s = {1, 2, 3, 3, 2}   # {1, 2, 3} — дубликаты удалены
s = set([1, 2, 3])     # из списка

# Изменение
s.add(4)
s.remove(3)            # ошибка если нет
s.discard(3)           # тихо если нет

# Операции с множествами
a = {1, 2, 3, 4}
b = {3, 4, 5, 6}

a & b           # {3, 4}       — пересечение
a | b           # {1,2,3,4,5,6}— объединение
a - b           # {1, 2}       — разность
a ^ b           # {1,2,5,6}    — симметричная разность

# Проверка
3 in s          # O(1) — быстро!
"x" in list     # O(n) — медленно

Когда что использовать

Коллекция Упорядочена Mutable Дубликаты O(in) Используй когда
list да да да O(n) упорядоченный набор, изменяется
tuple да нет да O(n) фиксированный набор, ключ dict
set нет да нет O(1) уникальность, быстрые проверки

enumerate, zip, range — встроенные итераторы

# enumerate — индекс + значение
for i, name in enumerate(names):           # i: 0,1,2...
    print(f"{i}. {name}")

for i, name in enumerate(names, start=1):  # i: 1,2,3...
    print(f"{i}. {name}")

# zip — параллельная итерация
names = ["Alice", "Bob"]
ages  = [25, 30]
for n, a in zip(names, ages):
    print(n, a)

dict(zip(names, ages))           # {"Alice": 25, "Bob": 30}

# zip останавливается на самом коротком
zip([1,2,3], [10, 20])           # пары: (1,10), (2,20)

# range — диапазон
range(5)              # 0,1,2,3,4
range(2, 8)           # 2,3,4,5,6,7
range(0, 10, 2)       # 0,2,4,6,8 — шаг
range(10, 0, -1)      # 10,9,8...1 — обратный

collections — готовые специализированные структуры

from collections import Counter, defaultdict, deque

# Counter — подсчёт частот в одной строке
c = Counter(["apple", "banana", "apple", "cherry"])
# Counter({"apple": 2, "banana": 1, "cherry": 1})

c.most_common(2)                 # [("apple", 2), ("banana", 1)] — топ N
c["apple"]                       # 2
c["unknown"]                     # 0 — нет KeyError

# defaultdict — dict с автоматическим дефолтом для новых ключей
groups = defaultdict(list)
for item in items:
    groups[item.category].append(item)
# Не нужно: if key not in groups: groups[key] = []

counts = defaultdict(int)        # дефолт 0
for word in words:
    counts[word] += 1            # сразу инкремент без get

# deque — двусторонняя очередь (O(1) с обоих концов)
from collections import deque
q = deque([1, 2, 3])
q.append(4)              # справа
q.appendleft(0)          # слева
q.pop()                  # справа
q.popleft()              # слева — у list это O(n)!

Counter заменяет ручной подсчёт через dict. defaultdict убирает проверки if key in dict. deque — когда нужно частое удаление слева.

Примеры

# Уникальные элементы из списка — через set
unique = list(set(words))

# Общие элементы двух списков — O(n)
common = list(set(lst1) & set(lst2))

# Топ-3 игрока по score
top3 = sorted(players, key=lambda p: p["score"], reverse=True)[:3]
names = [p["name"] for p in top3]

# Самое частое слово
from collections import Counter
most_common = Counter(words).most_common(1)[0][0]

# Группировка по полю
from collections import defaultdict
by_city = defaultdict(list)
for user in users:
    by_city[user["city"]].append(user["name"])

Слабые точки

  • list.sort() — сортирует на месте, возвращает None; sorted() — возвращает новый список
  • set не поддерживает индексацию: s[0] → TypeError
  • tuple как ключ dict работает только если все элементы hashable
  • Преобразование: list(s) — порядок НЕ гарантирован
