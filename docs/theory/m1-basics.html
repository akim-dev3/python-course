Циклы

# for по списку
for item in items:
    print(item)

# for по диапазону
for i in range(5):       # 0, 1, 2, 3, 4
    print(i)

for i in range(2, 10, 2):  # 2, 4, 6, 8
    print(i)

# while
while x > 0:
    x -= 1

# Управление
break       # выход из цикла
continue    # пропустить итерацию

# enumerate — индекс + значение
for i, val in enumerate(items):
    print(i, val)

Функции

# Базовое определение
def greet(name):
    return f"Hello, {name}!"

# Значения по умолчанию
def power(base, exp=2):
    return base ** exp

# Несколько возвращаемых значений (кортеж)
def min_max(lst):
    return min(lst), max(lst)

low, high = min_max([3, 1, 5])   # распаковка

# Именованные аргументы (kwargs при вызове)
power(base=3, exp=4)            # 81 — порядок не важен

# *args — переменное число позиционных
def total(*nums):
    return sum(nums)
total(1, 2, 3, 4)                # 10

# **kwargs — переменное число именованных
def build(**fields):
    return fields                # dict
build(name="Alice", age=25)      # {"name": "Alice", "age": 25}

# lambda — анонимная функция (одна строка)
square = lambda x: x ** 2
sorted(users, key=lambda u: u["age"])

Scope — область видимости

x = 10                  # global

def outer():
    y = 5               # local для outer
    def inner():
        nonlocal y      # доступ к y из outer
        y += 1
    inner()

def change():
    global x            # доступ к x вне функции
    x = 99

Внутри функции переменная — local по умолчанию. Чтобы изменить внешнюю — global (модуль) или nonlocal (вложенная функция).

f-строки

name = "Alice"
age = 25
print(f"{name} is {age} years old")
print(f"Score: {score:.2f}")    # 2 знака после запятой
print(f"{value:,}")              # 1,000,000 — разделители тысяч
print(f"{value:>10}")            # выравнивание справа в 10 символов
print(f"{value:0>3}")            # 007 — заполнение нулями

Обработка исключений

try:
    n = int(input("Число: "))
    result = 10 / n
except ValueError:
    print("Не число")
except ZeroDivisionError:
    print("Деление на ноль")
except (TypeError, KeyError) as e:
    print(f"Ошибка: {e}")
else:
    print("Без ошибок")          # если try прошёл
finally:
    print("Выполнится всегда")

Базовые исключения: ValueError, TypeError, KeyError, IndexError, ZeroDivisionError, FileNotFoundError.

match / case (Python 3.10+)

def http_status(code):
    match code:
        case 200:
            return "OK"
        case 404:
            return "Not Found"
        case 500 | 502 | 503:    # несколько значений
            return "Server Error"
        case _:                   # default
            return "Unknown"

Замена длинных if/elif/elif. Поддерживает распаковку структур (паттерн-матчинг).

Примеры

# Функция с условием и циклом
def categorize_scores(scores):
    result = {"A": 0, "B": 0, "C": 0}
    for score in scores:
        if score >= 90:
            result["A"] += 1
        elif score >= 70:
            result["B"] += 1
        else:
            result["C"] += 1
    return result
