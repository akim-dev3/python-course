Модуль 3 — Dict и агрегации

Конспект обновлён 2026-06-09

Суть

Dict-паттерны агрегации и работа с вложенными структурами list[dict]. Закрыт к 08.06 (drills/03_patterns p1–p5).

Детали

Dict — базовые операции

d = {"name": "Alice", "age": 25}

# Чтение
d["name"]                # "Alice" — KeyError если нет
d.get("age")             # 25 — None если нет
d.get("city", "—")       # "—" — дефолт если нет
"city" in d              # False — проверка ключа O(1)

# Запись
d["city"] = "Moscow"     # добавить / изменить
d.update({"age": 26, "country": "RU"})    # массовое обновление
d.setdefault("score", 0)                  # ставит если нет, иначе не трогает

# Удаление
del d["city"]                             # KeyError если нет
d.pop("city")                             # удалить и вернуть
d.pop("city", None)                       # безопасно с дефолтом
d.clear()                                 # очистить

# Итерация
for key in d:                # по ключам
for v in d.values():         # по значениям
for k, v in d.items():       # по парам

# Просмотр
d.keys()                     # dict_keys(['name', 'age'])
d.values()                   # dict_values(['Alice', 25])
d.items()                    # dict_items([('name', 'Alice'), ('age', 25)])

len(d)                       # количество пар

Dict Comprehension

# {ключ: значение for ... if ...}
squares = {x: x ** 2 for x in range(5)}
# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}

# Инверсия
inverted = {v: k for k, v in d.items()}

# Фильтрация
adults = {n: a for n, a in users.items() if a >= 18}

# Из двух списков
dict(zip(keys, values))

max / min с key — O(n) поиск экстремума

# Самый старший пользователь
oldest = max(users, key=lambda u: u["age"])

# Игрок с max score
winner = max(players, key=lambda p: p["score"])

# Ключ dict с максимальным значением
top_key = max(d, key=d.get)              # O(n) — быстрее sorted

# Несколько значений по ключу
sorted(users, key=lambda u: (u["age"], u["name"]))  # сначала age, потом name

Готовые инструменты из collections

from collections import Counter, defaultdict

# Counter — паттерн COUNT в одной строке
counts = Counter(p["city"] for p in users)
# Counter({"Moscow": 3, "SPb": 2})
counts.most_common(3)                    # топ-3

# defaultdict — паттерн GROUP BY без get(key, 0)
groups = defaultdict(list)
for u in users:
    groups[u["city"]].append(u["name"])

totals = defaultdict(int)
for o in orders:
    totals[o["user"]] += o["amount"]

Counter заменяет ручной d.get(k, 0) + 1. defaultdict убирает if key not in d и d.get(k, []). Используй, когда логика — чистый подсчёт или группировка.

ПАТТЕРН 1: COUNT

# Количество элементов, удовлетворяющих условию
sum(1 for x in data if условие)

# Пример: сколько premium-пользователей
count = sum(1 for p in players if p.get("premium"))

ПАТТЕРН 2: SUM

# Сумма поля с условием
sum(x.get("field", 0) for x in data if условие)

# Пример: суммарные монеты у premium-игроков
total = sum(p.get("coins", 0) for p in players if p.get("premium"))

ПАТТЕРН 3: FILTER (без dict!)

# Правило: если нет группировки — dict НЕ нужен
[x for x in data if условие]

# FILTER + SORTED → list[str]
result = [x for x in data if условие]
return [x["name"] for x in sorted(result, key=lambda x: x.get("field", 0), reverse=True)]

ПАТТЕРН 4: GROUP BY

# Когда один ключ встречается несколько раз → нужен dict
d = {}
for x in data:
    key = x.get("field")
    d[key] = d.get(key, 0) + value

# Пример: сколько заказов у каждого пользователя
orders_per_user = {}
for order in orders:
    user = order.get("user")
    orders_per_user[user] = orders_per_user.get(user, 0) + 1

ПАТТЕРН 5: AVG (всегда два dict)

totals = {}
counts = {}
for x in data:
    key = x.get("field")
    totals[key] = totals.get(key, 0) + x.get("value", 0)
    counts[key] = counts.get(key, 0) + 1
return {k: round(totals[k] / counts[k], 2) for k in totals}

ПАТТЕРН 6: TOP N

# Из dict → топ N ключей по значению
top = sorted(d.items(), key=lambda x: x[1], reverse=True)[:N]
return [k for k, v in top]

# MAX в dict
winner = max(d, key=d.get)   # O(n) — быстрее sorted

Ключевое правило

Есть группировка (один ключ встречается несколько раз) → dict
Нет группировки → FILTER + SORTED без dict

ВЛОЖЕННЫЕ СТРУКТУРЫ

Структура данных

orders = [
    {
        "id": 1,
        "customer": "Alice",
        "status": "paid",
        "items": [                             # вложенный список
            {"product": "Laptop", "price": 1200, "qty": 1},
            {"product": "Mouse",  "price": 25,   "qty": 2},
        ]
    },
    ...
]

Доступ к items

order.get("items", [])    # ВСЕГДА через get — защита от KeyError

ПАТТЕРН: any() / all() — условие на уровне заказа

# any() — ХОТЯ БЫ ОДИН товар дороже 1000
result = [
    order for order in orders
    if any(item["price"] > 1000 for item in order.get("items", []))
]

# all() — ВСЕ товары дешевле 1000
cheap = [
    order for order in orders
    if all(item["price"] < 1000 for item in order.get("items", []))
]

# ❌ ОШИБКА — теряем контекст заказа
result = [item for order in orders for item in order["items"] if item["price"] > 1000]

Слабая точка: any() / all() работают на уровне order, не item. Генератор внутри — это проверка условия для каждого item.

Особенности на пустых данных: - any([]) → False (нет элементов, удовлетворяющих условию) - all([]) → True (никто не нарушает условие — пустой набор всегда «все»)

ПАТТЕРН: агрегация по вложенным данным

# Сумма по одному заказу
def order_total(order):
    return sum(item["price"] * item.get("qty", 1) for item in order.get("items", []))

# Выручка по всем оплаченным
revenue = sum(order_total(o) for o in orders if o.get("status") == "paid")

# Разворачивание (все товары по всем заказам)
all_items = [item for order in orders for item in order.get("items", [])]

ПАТТЕРН: build_report

def build_report(orders):
    if not orders:
        return {}

    paid = [o for o in orders if o.get("status") == "paid"]
    totals = [order_total(o) for o in paid]

    return {
        "total_orders": len(orders),
        "paid_orders": len(paid),
        "revenue": sum(totals),
        "avg_order": sum(totals) / len(totals) if totals else 0,
        "big_orders": [
            o for o in orders
            if any(item["price"] > 1000 for item in o.get("items", []))
        ],
    }

Не защищаю order.get("items", []) Всегда .get("items", [])
