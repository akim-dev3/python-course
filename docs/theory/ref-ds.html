# ============================================================
# STACK (стек) — LIFO: Last In, First Out
# ============================================================
# В Python — обычный list. push = append, pop = pop (с конца, O(1)).
# Когда брать: нужно "откатиться назад" / проверить парность/вложенность /
# запомнить историю действий (undo).

stack = []
stack.append(1)      # push
stack.append(2)
top = stack[-1]       # peek, не удаляя — top == 2
stack.pop()            # pop -> 2, stack теперь [1]

# Частый шаблон — балансировка скобок:
pairs = {")": "(", "]": "[", "}": "{"}
# при встрече открывающей — push, при закрывающей — pop и сравнить с pairs[...]


# ============================================================
# QUEUE (очередь) — FIFO: First In, First Out
# ============================================================
# НЕ используем list.pop(0) — это O(n) (весь список сдвигается).
# Правильно — collections.deque: append/appendleft/pop/popleft — все O(1).

from collections import deque

q = deque()
q.append(1)          # enqueue (справа)
q.append(2)
first = q.popleft()   # dequeue (слева) -> 1

# ============================================================
# LINKED LIST (связный список) — свой класс, без готовой структуры в Python
# ============================================================
# Node хранит значение + ссылку на следующий узел. Список — цепочка Node.
# Когда брать в реальности: редко (list в Python и так хорош), но это
# классика собеседований — проверяет понимание ссылок/указателей.

class Node:
    def __init__(self, value):
        self.value = value
        self.next = None


class LinkedList:
    def __init__(self):
        self.head = None

    def push_front(self, value):
        node = Node(value)
        node.next = self.head
        self.head = node

    def to_list(self):
        result = []
        current = self.head
        while current:
            result.append(current.value)
            current = current.next
        return result

# ============================================================
# HASH TABLE "под капотом" — как устроен dict/set
# ============================================================
# dict хранит пары (key, value) в массиве корзин (buckets).
# 1. hash(key) даёт число
# 2. число % размер_массива даёт индекс корзины
# 3. коллизия (два ключа попали в одну корзину) — CPython цепляет
#    следующую свободную корзину (open addressing), не список поверх корзины
#
# Следствие: доступ/вставка/удаление — O(1) в среднем (амортизированно),
# но O(n) в худшем случае (много коллизий) — на практике почти не встречается.
# Ключ должен быть hashable (immutable: int, str, tuple — но не list/dict).

# Практическое применение хэширования помимо обычного dict — "видели ли мы это":
seen = set()
# if x in seen: ...   -> O(1), а не O(n) как в list
